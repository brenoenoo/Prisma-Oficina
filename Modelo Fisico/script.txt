CREATE DATABASE IF NOT EXISTS db_cliente_leticia;
USE db_cliente_leticia;

CREATE TABLE categoria (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL
);

CREATE TABLE configuracoes_loja (
    id_config_loja INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cnpj VARCHAR(18),
    telefone VARCHAR(20),
    email_comercial VARCHAR(150),
    endereco VARCHAR(200),
    cidade VARCHAR(100),
    estado VARCHAR(50),
    cep VARCHAR(9)
);

CREATE TABLE cliente (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    telefone VARCHAR(20),
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150)
);

CREATE TABLE usuario (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    ultimo_acesso DATETIME,
    senha VARCHAR(255) NOT NULL,
    tipo ENUM('gerente', 'funcionario') NOT NULL,
    email VARCHAR(150) NOT NULL,
    status ENUM('ativo', 'inativo') NOT NULL,
    nome VARCHAR(100) NOT NULL
);

CREATE TABLE produtos (
    id_produto INT AUTO_INCREMENT PRIMARY KEY,
    id_categoria INT NOT NULL,
    nome VARCHAR(100) NOT NULL,
    sku VARCHAR(50),
    preco DECIMAL(10,2) NOT NULL,
    estoque INT NOT NULL DEFAULT 0,
    descricao TEXT,
    capa VARCHAR(255),
    data_cadastro DATE NOT NULL,
    estoque_minimo INT NOT NULL DEFAULT 0,
    CONSTRAINT fk_produto_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria(id_categoria)
);

CREATE TABLE horario_funcionamento (
    id_horario INT AUTO_INCREMENT PRIMARY KEY,
    id_configuracoes_loja INT NOT NULL,
    dia_semana VARCHAR(20) NOT NULL,
    hora_abertura TIME NOT NULL,
    hora_fechamento TIME NOT NULL,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT fk_horario_configuracao
        FOREIGN KEY (id_configuracoes_loja)
        REFERENCES configuracoes_loja(id_config_loja)
);

CREATE TABLE agendamento (
    id_agendamento INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT NOT NULL,
    id_usuario INT NOT NULL,
    id_horario_funcionamento INT NOT NULL,
    data DATE NOT NULL,
    hora TIME NOT NULL,
    status ENUM('agendado', 'cancelado', 'concluido') NOT NULL,
    observacao TEXT,
    CONSTRAINT fk_agendamento_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),
    CONSTRAINT fk_agendamento_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuario(id_usuario),
    CONSTRAINT fk_agendamento_horario
        FOREIGN KEY (id_horario_funcionamento)
        REFERENCES horario_funcionamento(id_horario)
);

CREATE TABLE atendimento (
    id_atendimento INT AUTO_INCREMENT PRIMARY KEY,
    id_agendamento INT NOT NULL UNIQUE,
    id_cliente INT NOT NULL,
    id_usuario INT NOT NULL,
    data_hora DATETIME NOT NULL,
    descricao TEXT,
    status ENUM('aberto', 'concluido') NOT NULL,
    CONSTRAINT fk_atendimento_agendamento
        FOREIGN KEY (id_agendamento)
        REFERENCES agendamento(id_agendamento),
    CONSTRAINT fk_atendimento_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente),
    CONSTRAINT fk_atendimento_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuario(id_usuario)
);

CREATE TABLE registro_ponto (
    id_registro INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    data DATE NOT NULL,
    entrada TIME NOT NULL,
    saida TIME,
    CONSTRAINT fk_ponto_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuario(id_usuario)
);

CREATE TABLE movimentacao (
    id_movimentacao INT AUTO_INCREMENT PRIMARY KEY,
    id_produto INT NOT NULL,
    id_usuario INT NOT NULL,
    id_cliente INT,
    tipo ENUM('entrada', 'saida') NOT NULL,
    quantidade INT NOT NULL,
    estoque_anterior INT NOT NULL,
    estoque_atual INT NOT NULL,
    motivo VARCHAR(255),
    realizado_em DATETIME NOT NULL,
    CONSTRAINT fk_movimentacao_produto
        FOREIGN KEY (id_produto)
        REFERENCES produtos(id_produto),
    CONSTRAINT fk_movimentacao_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuario(id_usuario),
    CONSTRAINT fk_movimentacao_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES cliente(id_cliente)
);

CREATE TABLE relatorio (
    id_relatorio INT AUTO_INCREMENT PRIMARY KEY,
    periodo VARCHAR(50) NOT NULL,
    data_geracao DATETIME NOT NULL,
    tipo VARCHAR(50) NOT NULL
);

CREATE TABLE atendimento_relatorio (
    id_atendimento INT NOT NULL,
    id_relatorio INT NOT NULL,
    PRIMARY KEY (id_atendimento, id_relatorio),
    CONSTRAINT fk_ar_atendimento
        FOREIGN KEY (id_atendimento)
        REFERENCES atendimento(id_atendimento),
    CONSTRAINT fk_ar_relatorio
        FOREIGN KEY (id_relatorio)
        REFERENCES relatorio(id_relatorio)
);
